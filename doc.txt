## 6. Propostas recebidas + confirmação da escolha

**Ficheiro:** `lib/features/client/presentation/client_proposals_screen.dart`  
**Dependências novas:** nenhuma.

Inclui a lista de propostas, proposta recomendada, bottom sheet de confirmação, estado de loading bloqueado durante a aceitação, proposta expirada/retirada e erro com repetição sem fechar o sheet.

  accepted,

class ClientProposalAcceptResult {
  const ClientProposalAcceptResult({
    required this.outcome,
    this.errorMessage,
  });

  const ClientProposalAcceptResult.accepted()
      : outcome = ClientProposalAcceptOutcome.accepted,
        errorMessage = null;

  const ClientProposalAcceptResult.unavailable()
      : outcome = ClientProposalAcceptOutcome.unavailable,
        errorMessage = null;

  const ClientProposalAcceptResult.failed({
    this.errorMessage,
  }) : outcome = ClientProposalAcceptOutcome.failed;

  final ClientProposalAcceptOutcome outcome;
  final String? errorMessage;
}

class ClientProposalsScreen extends StatelessWidget {
  const ClientProposalsScreen({
    super.key,
    required this.dataAsync,
    required this.onBack,
    required this.onAcceptProposal,
    required this.onProposalAccepted,
    this.onOpenWorker,
    this.onRetry,
  });

  final AsyncValue<ClientProposalsViewData> dataAsync;

  final VoidCallback onBack;
  final VoidCallback? onRetry;

  final ValueChanged<String>? onOpenWorker;

  final Future<ClientProposalAcceptResult> Function(
    String proposalId,
  ) onAcceptProposal;

  /// Navegação pós-confirmação real.
  final ValueChanged<String> onProposalAccepted;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: AppColors.background,
      body: SafeArea(
        child: dataAsync.when(
          loading: () => _ProposalsLoading(
            onBack: onBack,
          ),
          error: (_, __) => _ProposalsError(
            onBack: onBack,
            onRetry: onRetry,
          ),
          data: (data) => _ProposalsContent(
            data: data,
            onBack: onBack,
            onOpenWorker: onOpenWorker,
            onChoose: (proposal) async {
              final accepted =
                  await _showAcceptProposalSheet(
                context: context,
                proposal: proposal,
                onAccept: onAcceptProposal,
              );

              if (accepted == true && context.mounted) {
                onProposalAccepted(proposal.proposalId);
              }
            },
          ),
        ),
      ),
    );
  }

  Future<bool?> _showAcceptProposalSheet({
    required BuildContext context,
    required ClientProposalViewData proposal,
    required Future<ClientProposalAcceptResult> Function(
      String proposalId,
    ) onAccept,
  }) {
    return showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,

      // Mantemos o sheet controlado pelo próprio conteúdo.
      // Assim nunca desaparece a meio de uma aceitação.
      isDismissible: false,
      enableDrag: false,

      backgroundColor: AppColors.surface,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(
          top: Radius.circular(AppRadius.card),
        ),
      ),
      builder: (_) => _AcceptProposalSheet(
        proposal: proposal,
        onAccept: onAccept,
      ),
    );
  }
}

enum _ProposalSheetState {
  confirmation,
  unavailable,
}

class _AcceptProposalSheet extends StatefulWidget {
  const _AcceptProposalSheet({
    required this.proposal,
    required this.onAccept,
  });

  final ClientProposalViewData proposal;

  final Future<ClientProposalAcceptResult> Function(
    String proposalId,
  ) onAccept;

  @override
  State<_AcceptProposalSheet> createState() =>
      _AcceptProposalSheetState();
}

class _AcceptProposalSheetState
    extends State<_AcceptProposalSheet> {
  bool _submitting = false;

  _ProposalSheetState _sheetState =
      _ProposalSheetState.confirmation;

  Future<void> _accept() async {
    if (_submitting) {
      return;
    }

    setState(() {
      _submitting = true;
    });

    ClientProposalAcceptResult result;

    try {
      result = await widget.onAccept(
        widget.proposal.proposalId,
      );
    } catch (_) {
      result = const ClientProposalAcceptResult.failed();
    }

    if (!mounted) {
      return;
    }

    switch (result.outcome) {
      case ClientProposalAcceptOutcome.accepted:
        Navigator.of(context).pop(true);
        return;

      case ClientProposalAcceptOutcome.unavailable:
        setState(() {
          _submitting = false;
          _sheetState =
              _ProposalSheetState.unavailable;
        });
        return;

      case ClientProposalAcceptOutcome.failed:
        setState(() {
          _submitting = false;
        });

        _showRetrySnackBar(
          result.errorMessage,
        );
        return;
    }
  }

  void _showRetrySnackBar(String? message) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(
            message ??
                'Não foi possível aceitar. Tente novamente.',
          ),
          action: SnackBarAction(
            label: 'Repetir',
            onPressed: _accept,
          ),
        ),
      );
  }

  @override
  Widget build(BuildContext context) {
    final keyboardInset =
        MediaQuery.viewInsetsOf(context).bottom;

    return PopScope(
      canPop: !_submitting,
      child: Padding(
        padding: EdgeInsets.fromLTRB(
          AppSpacing.md,
          AppSpacing.sm,
          AppSpacing.md,
          AppSpacing.md + keyboardInset,
        ),
        child: AppFadeThroughSwitcher(
          switchKey: _sheetState,
          duration: const Duration(milliseconds: 220),
          child:
              _sheetState == _ProposalSheetState.unavailable
                  ? _UnavailableProposal(
                      key: const ValueKey(
                        'proposal_unavailable',
                      ),
                      onBackToProposals: () {
                        Navigator.of(context).pop(false);
                      },
                    )
                  : _ProposalConfirmation(
                      key: const ValueKey(
                        'proposal_confirmation',
                      ),
                      proposal: widget.proposal,
                      submitting: _submitting,
                      onConfirm: _accept,
                      onBack: _submitting
                          ? null
                          : () {
                              Navigator.of(context)
                                  .pop(false);
                            },
                    ),
        ),
      ),
    );
  }
}

### Notas de integração

`isRecommended` é apenas destaque visual e **não cria estado novo**. Preço, rating, prazo, data, pagamento e serviços chegam prontos da integração.

O bottom sheet só termina com sucesso quando `onAcceptProposal()` devolve `accepted`. Se a proposta tiver expirado ou sido retirada, permanece no fluxo e mostra o estado próprio. Se houver falha de rede, o sheet mantém-se aberto e o `SnackBar` permite repetir.

Durante `"A confirmar..."`, o CTA e o botão de voltar ficam bloqueados. O modal também não permite swipe/tap exterior, evitando que desapareça enquanto existe uma operação em curso.

### Notas de integração

- O header recebe `jobStatusPresentation` do presenter oficial. **Não foi criado um `JobStatus.agendado`** só porque o mock mostra “Agendado”; o Tech Lead mapeia o modelo real existente.
- O `StatusTimeline` também é injetado já configurado, usando os tokens oficiais do sistema de estados.
- Datas, horas, moradas e valores acordados chegam já formatados.
- `canReschedule` e `canCancel` vêm das regras reais de domínio; o ecrã não decide quando estas ações são permitidas.
- **Remarcar** apenas dispara `onReschedule`. O fluxo completo de `RescheduleStatus` continua separado e usa os presenters definitivos.
- O botão principal **Enviar mensagem** e o ícone de contacto abrem exclusivamente WhatsApp. Não existe ação de chamada.
- Também aqui foi omitido o selo de identidade junto ao nome por não existir essa funcionalidade real.
- `AppStaggeredEntrance` é aplicado aos blocos; loading usa exclusivamente `AppSkeletonShimmer`.
